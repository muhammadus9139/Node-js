# Open MongoDB Shell

mongosh


# ======================
# Database Operations
# ======================

# Show all databases

show dbs


# Create / Switch Database

use studentDB


# Check Current Database

db


# Drop Database

db.dropDatabase()



# ======================
# Collection Operations
# ======================

# Create Collection

db.createCollection("students")


# Show Collections

show collections



# ======================
# Insert Data
# ======================

# Insert One Document

db.students.insertOne({
    name:"Ali",
    age:20,
    department:"CS"
})


# Insert Multiple Documents

db.students.insertMany([
    {
        name:"Ahmed",
        age:22,
        department:"SE"
    },
    {
        name:"Sara",
        age:21,
        department:"CS"
    }
])



# ======================
# Read Data
# ======================

# Show All Data

db.students.find()


# Pretty Format

db.students.find().pretty()



# ======================
# Delete Data
# ======================

# Delete One

db.students.deleteOne({
    name:"Ali"
})


# Delete Multiple

db.students.deleteMany({
    age:21
})



# ======================
# Delete Collection
# ======================

db.students.drop()
